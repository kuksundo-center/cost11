내 구글 드라이브(구글 시트) 실시간 자동 저장 설정

구글 드라이브에 만든 구글 시트에 무료 스크립트(Apps Script)를 1회 배포하고 URL을 등록하면, 웹사이트에서 입력 또는 삭제할 때마다 내 구글 시트에 실시간 반영됩니다.

※ URL을 입력하고 [연동 저장]을 누르면 즉시 실시간 동기화가 시작됩니다.

구글 시트에 붙여넣을 스크립트 코드 (입력 & 삭제 & 조회 지원)

구글 시트 상단 메뉴 [확장 프로그램] → [Apps Script]를 누른 후 기존 코드를 지우고 아래 코드를 붙여넣은 뒤, [배포] → [새 배포] → 웹 앱 (액세스 권한: 모든 사용자)으로 배포하세요. 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var records = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      records.push({
        id: rows[i][0],
        date: formatDate(rows[i][1]),
        type: rows[i][2] === '수입' ? 'income' : 'expense',
        category: rows[i][3] || '',
        title: rows[i][4] || '',
        expenseAmount: Number(rows[i][5]) || 0,
        incomeAmount: Number(rows[i][6]) || 0,
        note: rows[i][7] || ''
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(records))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "날짜", "구분", "카테고리", "항목내용", "비용금액", "수입금액", "비고", "등록일시"]);
  }

  if (data.action === 'delete') {
    var values = sheet.getDataRange().getValues();
    for (var i = values.length - 1; i >= 1; i--) {
      if (String(values[i][0]) === String(data.id)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'deleted'}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  sheet.appendRow([
    data.id,
    data.date,
    data.type === 'income' ? '수입' : '비용',
    data.category,
    data.title,
    data.expenseAmount,
    data.incomeAmount,
    data.note,
    new Date()
  ]);

  return ContentService.createTextOutput(JSON.stringify({status: 'success'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function formatDate(val) {
  if (val instanceof Date) {
    var y = val.getFullYear();
    var m = ('0' + (val.getMonth() + 1)).slice(-2);
    var d = ('0' + val.getDate()).slice(-2);
    return y + '-' + m + '-' + d;
  }
  return String(val);
}

GitHub Pages로 전 세계 무료 배포하기

1. 파일 이름 저장

현재 코드를 컴퓨터에 저장할 때 반드시 index.html 로 저장하세요.

2. GitHub Repository 생성

GitHub 로그인 → [New repository] → 저장소명(예: ledger) 입력 후 Public 선택하고 생성합니다.

3. 파일 업로드 및 Pages 활성화

저장소에 index.html 파일을 업로드한 뒤, 저장소 상단 [Settings] → [Pages] 로 이동합니다. Branch를 main으로 바꾸고 Save를 누르면 끝납니다.
