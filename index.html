function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateLedgerSheet(ss);
  var rows = sheet.getDataRange().getValues();
  var records = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      records.push({
        id: rows[i][0],
        date: formatDate(rows[i][1]),
        type: rows[i][2] === '수입' ? 'income' : 'expense',
        category: rows[i][3] || '',
        title: rows[i][4] || '',
        expenseAmount: Number(rows[i][5]) || 0,
        incomeAmount: Number(rows[i][6]) || 0,
        note: rows[i][7] || ''
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(records)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ledgerSheet = getOrCreateLedgerSheet(ss);
  var data = JSON.parse(e.postData.contents);

  if (data.action === 'delete') {
    var values = ledgerSheet.getDataRange().getValues();
    for (var i = values.length - 1; i >= 1; i--) {
      if (String(values[i][0]) === String(data.id)) {
        ledgerSheet.deleteRow(i + 1);
        break;
      }
    }
    updateMonthlySummarySheet(ss, ledgerSheet);
    return ContentService.createTextOutput(JSON.stringify({status: 'deleted'})).setMimeType(ContentService.MimeType.JSON);
  }

  ledgerSheet.appendRow([
    data.id,
    data.date,
    data.type === 'income' ? '수입' : '비용',
    data.category,
    data.title,
    Number(data.expenseAmount) || 0,
    Number(data.incomeAmount) || 0,
    data.note || '',
    new Date()
  ]);

  var lastRow = ledgerSheet.getLastRow();
  ledgerSheet.getRange(lastRow, 6, 1, 2).setNumberFormat("#,##0");
  ledgerSheet.getRange(lastRow, 2).setNumberFormat("yyyy-mm-dd");

  updateMonthlySummarySheet(ss, ledgerSheet);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateLedgerSheet(ss) {
  var sheet = ss.getSheetByName("거래원장");
  if (!sheet) {
    sheet = ss.insertSheet("거래원장", 0);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "날짜", "구분", "카테고리", "항목내용", "비용금액", "수입금액", "비고", "등록일시"]);
    sheet.getRange(1, 1, 1, 9).setBackground("#047857").setFontColor("#ffffff").setFontWeight("bold").setHorizontalAlignment("center");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function updateMonthlySummarySheet(ss, ledgerSheet) {
  var summarySheet = ss.getSheetByName("월별자동집계");
  if (!summarySheet) {
    summarySheet = ss.insertSheet("월별자동집계", 1);
  }
  summarySheet.clear();
  summarySheet.appendRow(["연월", "총 수입", "총 비용", "순이익", "건수"]);
  summarySheet.getRange(1, 1, 1, 5).setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold").setHorizontalAlignment("center");
  summarySheet.setFrozenRows(1);

  var rows = ledgerSheet.getDataRange().getValues();
  var monthlyMap = {};
  var totalInc = 0, totalExp = 0;

  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    var dStr = formatDate(rows[i][1]).substring(0, 7);
    var exp = Number(rows[i][5]) || 0;
    var inc = Number(rows[i][6]) || 0;
    totalExp += exp; totalInc += inc;
    if (!monthlyMap[dStr]) monthlyMap[dStr] = { inc: 0, exp: 0, cnt: 0 };
    monthlyMap[dStr].inc += inc;
    monthlyMap[dStr].exp += exp;
    monthlyMap[dStr].cnt += 1;
  }

  var keys = Object.keys(monthlyMap).sort();
  var out = [];
  for (var j = 0; j < keys.length; j++) {
    var k = keys[j];
    out.push([k, monthlyMap[k].inc, monthlyMap[k].exp, monthlyMap[k].inc - monthlyMap[k].exp, monthlyMap[k].cnt]);
  }
  if (out.length > 0) {
    summarySheet.getRange(2, 1, out.length, 5).setValues(out);
    summarySheet.getRange(2, 2, out.length, 3).setNumberFormat("#,##0");
    var lastRow = out.length + 2;
    summarySheet.getRange(lastRow, 1, 1, 5).setValues([["전체 누적 합계", totalInc, totalExp, totalInc - totalExp, rows.length - 1]]);
    summarySheet.getRange(lastRow, 1, 1, 5).setBackground("#f1f5f9").setFontWeight("bold");
    summarySheet.getRange(lastRow, 2, 1, 3).setNumberFormat("#,##0");
  }
}

function formatDate(val) {
  if (val instanceof Date) {
    var y = val.getFullYear();
    var m = ('0' + (val.getMonth() + 1)).slice(-2);
    var d = ('0' + val.getDate()).slice(-2);
    return y + '-' + m + '-' + d;
  }
  return String(val);
}
